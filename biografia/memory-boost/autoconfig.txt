        function drawMatrix() {
            ctx.fillStyle = 'rgba(0, 0, 0, 0.05)';
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            ctx.fillStyle = '#00ff00';
            ctx.font = fontSize + 'px monospace';

            for (let i = 0; i < drops.length; i++) {
                const text = chars[Math.floor(Math.random() * chars.length)];
                ctx.fillText(text, i * fontSize, drops[i] * fontSize);

                if (drops[i] * fontSize > canvas.height && Math.random() > 0.975) {
                    drops[i] = 0;
                }
                drops[i]++;
            }
        }

        setInterval(drawMatrix, 35);

        function analyzeFile(event) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                const content = e.target.result;
                log('readingLog', `Arquivo carregado: ${file.name} (${file.size} bytes)`, 'success');
                
                const analysis = analyzeContent(content, file.name);
                displayAnalysis(analysis);
            };
            reader.readAsText(file);
        }

        function analyzeContent(content, filename) {
            const analysis = {
                filename,
                size: content.length,
                type: filename.split('.').pop(),
                lines: content.split('\n').length,
                words: content.split(/\s+/).length,
                patterns: [],
                keywords: []
            };

            // Detectar padrões
            const patterns = [
                { name: 'Protocolo CRYPY', regex: /CRYPY|crypy/gi },
                { name: 'Referência a Sensei', regex: /Sensei|S3nsei|Criador/gi },
                { name: 'Datas', regex: /\d{2}\/\d{2}\/\d{4}/g },
                { name: 'Versões', regex: /V\d+|v\d+\.\d+/gi },
                { name: 'Regras/Leis', regex: /Regra|Lei|Protocolo/gi },
                { name: 'Segurança', regex: /segurança|proteção|backup/gi },
                { name: 'NFT/Cripto', regex: /NFT|token|blockchain|web3/gi }
            ];

            patterns.forEach(p => {
                const matches = content.match(p.regex);
                if (matches) {
                    analysis.patterns.push({ name: p.name, count: matches.length });
                }
            });

            // Extrair palavras-chave
            const commonWords = ['de', 'a', 'o', 'que', 'e', 'do', 'da', 'em', 'um', 'para', 'com', 'não', 'por', 'mais', 'as', 'os', 'no', 'se', 'na', 'eu', 'lhe', 'me', 'te', 'nos', 'vos', 'lhes', 'meu', 'minha', 'meus', 'minhas', 'seu', 'sua', 'seus', 'suas', 'nosso', 'nossa', 'nossos', 'nossas', 'este', 'esta', 'estes', 'estas', 'esse', 'essa', 'esses', 'essas', 'aquele', 'aquela', 'aqueles', 'aquelas', 'isto', 'isso', 'aquilo', 'qual', 'quais', 'quem', 'que', 'quanto', 'quanta', 'quantos', 'quantas', 'como', 'quando', 'onde', 'porque', 'porquê', 'sim', 'não', 'talvez', 'provavelmente', 'certamente', 'realmente', 'verdadeiramente', 'efetivamente', 'de fato', 'de verdade'];
            
            const words = content.toLowerCase().split(/\s+/);
            const wordCount = {};
            
            words.forEach(word => {
                word = word.replace(/[^\wáéíóúâêôãõç]/gi, '');
                if (word.length > 3 && !commonWords.includes(word)) {
                    wordCount[word] = (wordCount[word] || 0) + 1;
                }
            });

            analysis.keywords = Object.entries(wordCount)
                .sort((a, b) => b[1] - a[1])
                .slice(0, 10)
                .map(([word, count]) => ({ word, count }));

            return analysis;
        }

        function displayAnalysis(analysis) {
            let html = `<strong>📄 Arquivo:</strong> ${analysis.filename}<br>`;
            html += `<strong>📊 Tamanho:</strong> ${analysis.size} bytes | ${analysis.lines} linhas | ${analysis.words} palavras<br>`;
            html += `<strong>🏷️ Tipo:</strong> ${analysis.type}<br><br>`;
            
            if (analysis.patterns.length > 0) {
                html += `<strong>🎯 Padrões Detectados:</strong><br>`;
                analysis.patterns.forEach(p => {
                    html += `• ${p.name}: ${p.count} ocorrências<br>`;
                });
            }

            if (analysis.keywords.length > 0) {
                html += `<br><strong>🔑 Palavras-chave:</strong><br>`;
                analysis.keywords.forEach(k => {
                    html += `• ${k.word} (${k.count}x)<br>`;
                });
            }

            document.getElementById('patternList').innerHTML = html;
            log('readingLog', `Análise completa: ${analysis.patterns.length} padrões, ${analysis.keywords.length} palavras-chave`, 'success');
            
            // Registrar aprendizado
            systemState.patterns.push(...analysis.patterns);
            updateStats();
        }

        function extractPatterns() {
            const text = document.getElementById('textInput').value;
            if (!text.trim()) {
                log('readingLog', '⚠️ Nenhum texto para extrair padrões.', 'warning');
                return;
            }

            // Extrair padrões estruturais
            const patterns = {
                datas: text.match(/\d{2}\/\d{2}\/\d{4}/g) || [],
                versoes: text.match(/V\d+|v\d+\.\d+/gi) || [],
                emails: text.match(/[\w.-]+@[\w.-]+\.\w+/g) || [],
                urls: text.match(/https?:\/\/[^\s]+/g) || [],
                codigos: text.match(/[A-Z]{2,}-\d{3,}/g) || []
            };

            let html = '<strong>🎯 Padrões Estruturais Extraídos:</strong><br>';
            Object.entries(patterns).forEach(([type, matches]) => {
                if (matches.length > 0) {
                    html += `<br><strong>${type}:</strong> ${matches.length} encontrados<br>`;
                    matches.slice(0, 5).forEach(m => {
                        html += `• ${m}<br>`;
                    });
                }
            });

            document.getElementById('patternList').innerHTML = html;
            log('readingLog', `Padrões extraídos: ${Object.values(patterns).reduce((a, b) => a + b.length, 0)} total`, 'success');
        }
